// Service
@Service
@Transactional
public class SuperCostService {
    
    @Autowired
    private SuperCostRepository repository;
    
    @Transactional
    public int annulerDernierSuperCost() {
        List<SuperCost> lastCosts = repository.getLastSuperCost();
        int updatedCount = 0;
        
        for (SuperCost cost : lastCosts) {
            cost.setCout(0.0);
            repository.save(cost);  // ← IMPORTANT: Sauvegarder en base
            updatedCount++;
        }
        
        return updatedCount;
    }
}

// Controller
@PostMapping("/annuler/last")
public ResponseEntity<?> annulerDernierSuperCost() {
    int count = service.annulerDernierSuperCost();
    return ResponseEntity.ok(Map.of(
        "message", "Derniers coûts annulés avec succès",
        "nombre_modifies", count
    ));
}
